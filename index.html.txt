<script>
(function(){
  const STORAGE_KEY = 'bigas-ledger-entries';
  let entries = [];
  let currentType = 'purchase';
  let editingId = null;
  let activeFilter = 'all';
  let loaded = false;

  const $ = id => document.getElementById(id);
  const peso = n => '₱' + (Math.round(n*100)/100).toLocaleString('en-PH', {minimumFractionDigits: 0, maximumFractionDigits: 2});

  function fmtDate(d){
    const dt = new Date(d + 'T00:00:00');
    return dt.toLocaleDateString('en-PH', {month:'short', day:'numeric', year:'numeric'});
  }

  async function loadEntries(){
    try{
      const stored = localStorage.getItem(STORAGE_KEY);
      entries = stored ? JSON.parse(stored) : [];
    }catch(e){
      entries = [];
    }
    loaded = true;
    render();
  }

  async function persist(){
    try{
      localStorage.setItem(STORAGE_KEY, JSON.stringify(entries));
    }catch(e){
      showToast('Could not save — try again');
    }
  }

  function showToast(msg){
    const t = $('toast');
    t.textContent = msg;
    t.classList.add('show');
    setTimeout(()=> t.classList.remove('show'), 1600);
  }

  function computeSummary(){
    let revenue=0, expense=0, kgBought=0, kgSold=0, costKgTotal=0;
    entries.forEach(e=>{
      if(e.type==='sale'){ revenue += e.total; kgSold += e.qty; }
      else { expense += e.total; kgBought += e.qty; costKgTotal += e.total; }
    });
    const profit = revenue - expense;
    const stock = kgBought - kgSold;
    const avgCost = kgBought > 0 ? costKgTotal / kgBought : 0;
    return {revenue, expense, profit, stock, avgCost};
  }

  function renderStub(e){
    const isSale = e.type === 'sale';
    return `
      <div class="stub">
        <div class="stub-tab ${e.type}"></div>
        <div class="stub-body">
          <div class="stub-top">
            <div>
              <div class="stub-type ${e.type}">${isSale ? 'Sold' : 'Bought'}</div>
              <div class="stub-party">${escapeHtml(e.party || (isSale ? 'Customer' : 'Supplier'))}</div>
              <div class="stub-date">${fmtDate(e.date)}</div>
            </div>
            <div>
              <div class="stub-amt" style="color:${isSale ? 'var(--profit)' : 'var(--rust)'}">${isSale?'+':'-'}${peso(e.total)}</div>
              <div class="stub-qty">${e.qty} kg @ ₱${e.price}/kg</div>
            </div>
          </div>
          ${e.note ? `<div class="stub-note">${escapeHtml(e.note)}</div>` : ''}
          <div class="stub-del" data-del="${e.id}">Delete</div>
        </div>
      </div>`;
  }

  function escapeHtml(s){
    const d = document.createElement('div');
    d.textContent = s;
    return d.innerHTML;
  }

  function render(){
    const s = computeSummary();
    $('netAmt').textContent = Math.round(Math.abs(s.profit)).toLocaleString('en-PH');
    $('netAmt').parentElement.querySelector('.cur').textContent = s.profit < 0 ? '-₱' : '₱';
    $('netCaption').textContent = entries.length === 0
      ? 'No entries yet — add your first sale or purchase'
      : (s.profit >= 0 ? 'You are in the green' : 'Currently operating at a loss');

    $('statRevenue').textContent = peso(s.revenue);
    $('statExpense').textContent = peso(s.expense);
    $('statStock').textContent = (Math.round(s.stock*10)/10) + ' kg';
    $('statAvgCost').textContent = peso(s.avgCost);

    if(!loaded){
      $('listRecent').innerHTML = '<div class="loading-note">Loading your ledger…</div>';
      $('listAll').innerHTML = '';
      return;
    }

    const sorted = [...entries].sort((a,b)=> b.date.localeCompare(a.date) || b.id.localeCompare(a.id));

    if(sorted.length === 0){
      const empty = `<div class="empty-state"><div class="em-icon">🌾</div><p>Nothing logged yet.<br>Tap "Add entry" to record your first rice purchase or sale.</p></div>`;
      $('listRecent').innerHTML = empty;
      $('listAll').innerHTML = empty;
    } else {
      $('listRecent').innerHTML = `<div id="list">${sorted.slice(0,5).map(renderStub).join('')}</div>`;
      const filtered = activeFilter === 'all' ? sorted : sorted.filter(e => e.type === activeFilter);
      $('listAll').innerHTML = filtered.length
        ? `<div id="list">${filtered.map(renderStub).join('')}</div>`
        : `<div class="empty-state"><p>No ${activeFilter === 'purchase' ? 'purchases' : 'sales'} logged yet.</p></div>`;
    }

    document.querySelectorAll('[data-del]').forEach(el=>{
      el.onclick = () => deleteEntry(el.getAttribute('data-del'));
    });
  }

  async function deleteEntry(id){
    entries = entries.filter(e => e.id !== id);
    render();
    await persist();
    showToast('Entry deleted');
  }

  document.querySelectorAll('.tab-btn').forEach(btn=>{
    btn.onclick = () => {
      document.querySelectorAll('.tab-btn').forEach(b=>b.classList.remove('active'));
      btn.classList.add('active');
      const tab = btn.getAttribute('data-tab');
      $('dashTab').style.display = tab === 'dashboard' ? '' : 'none';
      $('entriesTab').style.display = tab === 'entries' ? '' : 'none';
    };
  });

  document.querySelectorAll('.pill').forEach(p=>{
    p.onclick = () => {
      document.querySelectorAll('.pill').forEach(x=>x.classList.remove('active'));
      p.classList.add('active');
      activeFilter = p.getAttribute('data-filter');
      render();
    };
  });

  function openSheet(){
    editingId = null;
    $('sheetTitle').textContent = 'New entry';
    $('fParty').value = '';
    $('fQty').value = '';
    $('fPrice').value = '';
    $('fNote').value = '';
    $('fDate').value = new Date().toISOString().slice(0,10);
    setType('purchase');
    updateTotalPreview();
    $('sheetOverlay').classList.add('open');
  }
  function closeSheet(){ $('sheetOverlay').classList.remove('open'); }

  $('openSheetBtn').onclick = openSheet;
  $('cancelBtn').onclick = closeSheet;
  $('sheetOverlay').onclick = (e) => { if(e.target === $('sheetOverlay')) closeSheet(); };

  function setType(t){
    currentType = t;
    $('optPurchase').className = 'type-opt' + (t==='purchase' ? ' sel-purchase' : '');
    $('optSale').className = 'type-opt' + (t==='sale' ? ' sel-sale' : '');
    $('fParty').placeholder = t === 'sale' ? 'e.g. Mang Tomas Sari-sari Store' : 'e.g. Aling Nena, Del Monte Traders';
    $('tpLabel').textContent = t === 'sale' ? 'Total revenue' : 'Total cost';
    updateTotalPreview();
  }
  $('optPurchase').onclick = () => setType('purchase');
  $('optSale').onclick = () => setType('sale');

  document.querySelectorAll('.qchip').forEach(chip=>{
    chip.onclick = () => {
      const add = parseFloat(chip.getAttribute('data-add'));
      const cur = parseFloat($('fQty').value) || 0;
      $('fQty').value = cur + add;
      updateTotalPreview();
    };
  });

  $('fQty').oninput = updateTotalPreview;
  $('fPrice').oninput = updateTotalPreview;

  function updateTotalPreview(){
    const qty = parseFloat($('fQty').value) || 0;
    const price = parseFloat($('fPrice').value) || 0;
    $('tpVal').textContent = peso(qty * price);
    $('saveBtn').disabled = !(qty > 0 && price > 0);
  }

  $('saveBtn').onclick = async () => {
    const qty = parseFloat($('fQty').value) || 0;
    const price = parseFloat($('fPrice').value) || 0;
    if(qty <= 0 || price <= 0) return;

    const entry = {
      id: editingId || (Date.now().toString(36) + Math.random().toString(36).slice(2,7)),
      type: currentType,
      party: $('fParty').value.trim(),
      qty, price,
      total: Math.round(qty*price*100)/100,
      date: $('fDate').value || new Date().toISOString().slice(0,10),
      note: $('fNote').value.trim()
    };

    if(editingId){
      entries = entries.map(e => e.id === editingId ? entry : e);
    } else {
      entries.push(entry);
    }
    closeSheet();
    render();
    await persist();
    showToast(currentType === 'sale' ? 'Sale recorded' : 'Purchase recorded');
  };
